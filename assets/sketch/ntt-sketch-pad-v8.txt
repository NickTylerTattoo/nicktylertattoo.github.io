  function loadTemplate(key, done) {
    if (!key) return;

    (WAITERS[key] = WAITERS[key] || []).push(done);
    if (IMGS[key]) return;

    im.crossOrigin = 'anonymous';   /* keeps the canvas exportable */
    im.onload = function () { READY[key] = true; flush(key); };
    im.onerror = function () { FAILED[key] = true; flush(key); };
    im.src = TEMPLATE_BASE + key + '.jpg?v=' + TEMPLATE_VERSION;
    IMGS[key] = im;
  }

    c.imageSmoothingEnabled = true;
    try { c.imageSmoothingQuality = 'high'; } catch (_) {}
    c.drawImage(IMGS[key], 0, 0, w, h);
    c.restore();

      mctx.font = '500 ' + fs + 'px ' + CFG.font;
      var tw = s.text ? mctx.measureText(s.text).width : fs * 0.6;
      var max = w * 0.94;
      if (tw > max) { fs = fs * max / tw; tw = max; }
      var m = w * 0.02;
      var cx = Math.max(m, Math.min(s.x * w, w - m));
      var cy = Math.max(m, Math.min(s.y * w, w * CFG.aspect - m));
      return { fs: fs, cx: cx, cy: cy, w: tw, h: fs * 1.3, a: (s.r || 0) * Math.PI / 180 };
    }
    function drawStroke(c, s, w) {
      if (s.t === 'text') {
        if (!s.text) return;
        var L = textLayout(s, w);
        c.save();
        c.translate(L.cx, L.cy);
        if (L.a) c.rotate(L.a);
        c.font = '500 ' + L.fs + 'px ' + CFG.font;
        c.textBaseline = 'middle'; c.textAlign = 'center';
        c.fillStyle = CFG.ink;
        c.fillText(s.text, 0, 0);
        c.restore();
        return;
      }
      var mult = s.t === 'shade' ? 3.2 : s.t === 'erase' ? 2.6 : 1;
      var lw = s.w / CFG.basePx * w * mult;
      var col = s.t === 'shade' ? CFG.shade : CFG.ink;
      var p = s.p, n = p.length / 2, i;
      if (!n) return;
      c.save();
      if (s.t === 'erase') c.globalCompositeOperation = 'destination-out';
      c.lineCap = 'round'; c.lineJoin = 'round'; c.lineWidth = lw;
      c.strokeStyle = col; c.fillStyle = col;
      c.beginPath();
      if (n === 1) {
        c.arc(p[0] * w, p[1] * w, lw / 2, 0, Math.PI * 2); c.fill();
      } else {
        c.moveTo(p[0] * w, p[1] * w);
        for (i = 1; i < n - 1; i++) {
          var x = p[i * 2] * w, y = p[i * 2 + 1] * w, nx = p[i * 2 + 2] * w, ny = p[i * 2 + 3] * w;
          c.quadraticCurveTo(x, y, (x + nx) / 2, (y + ny) / 2);
        }
        c.lineTo(p[(n - 1) * 2] * w, p[(n - 1) * 2 + 1] * w);
        c.stroke();
      }
      c.restore();
    }
    function buildUnder() {
      uctx.fillStyle = CFG.paper; uctx.fillRect(0, 0, under.width, under.height);
      drawBody(uctx, under.width, under.height, fileKey());
    }
    function redrawBase() {
      bctx.clearRect(0, 0, base.width, base.height);
      for (var i = 0; i < strokes.length; i++) drawStroke(bctx, strokes[i], W);
    }
    function present() {
      if (raf) { (window.cancelAnimationFrame || clearTimeout)(raf); raf = 0; }
      ctx.drawImage(under, 0, 0);
      if (cur && cur.t === 'erase') {
        lctx.clearRect(0, 0, live.width, live.height);
        lctx.drawImage(base, 0, 0);
        drawStroke(lctx, cur, W);
        ctx.drawImage(live, 0, 0);
      } else {
        ctx.drawImage(base, 0, 0);
        if (cur) drawStroke(ctx, cur, W);
      }
      if (float) {
        drawStroke(ctx, float.item, W);
        if (float.editing || drag) {
          var L = textLayout(float.item, W), pd = W * 0.012;
          ctx.save();
          ctx.translate(L.cx, L.cy);
          if (L.a) ctx.rotate(L.a);
          if (ctx.setLineDash) ctx.setLineDash([W * 0.012, W * 0.008]);
          ctx.lineWidth = Math.max(1, W / 460);
          ctx.strokeStyle = 'rgba(17,17,17,0.45)';
          ctx.strokeRect(-L.w / 2 - pd, -L.h / 2 - pd / 2, L.w + pd * 2, L.h + pd);
          ctx.restore();
        }
      }
    }
    function schedule() {
      if (raf) return;
      raf = (window.requestAnimationFrame || function (f) { return setTimeout(f, 16); })(function () { raf = 0; present(); });
    }
    function fit() {
      var r = stage.getBoundingClientRect();
      if (r.width < 10) return;
      var dpr = Math.min(window.devicePixelRatio || 1, 3);
      var w = Math.round(r.width * dpr), h = Math.round(w * CFG.aspect);
      if (cv.width === w && W === w) return;
      cv.width = under.width = base.width = live.width = w;
      cv.height = under.height = base.height = live.height = h;
      W = w;
      buildUnder(); redrawBase(); present();
    }
    if (window.ResizeObserver) new ResizeObserver(fit).observe(stage);
    else window.addEventListener('resize', fit);
    fit(); (window.requestAnimationFrame || setTimeout)(fit);

    /* ---------- pointer input ---------- */
    function norm(e) { return { x: (e.clientX - rect.left) / rect.width, y: (e.clientY - rect.top) / rect.width }; }
    function addPt(q, raw) {
      var x = q.x, y = q.y;
      if (!raw && cur.t !== 'erase' && CFG.smoothing > 0) {
        var a = 1 - Math.min(CFG.smoothing, 0.9);
        cur.fx += (x - cur.fx) * a; cur.fy += (y - cur.fy) * a;
        x = cur.fx; y = cur.fy;
      }
      var p = cur.p, n = p.length;
      if (n >= 2) { var dx = x - p[n - 2], dy = y - p[n - 1]; if (dx * dx + dy * dy < 0.0000064) return; }
      p.push(r4(x), r4(y));
    }
    function relax(p) {
      var n = p.length / 2;
      if (n < 4) return p;
      var o = p.slice();
      for (var i = 1; i < n - 1; i++) {
        o[i * 2] = r4(p[i * 2 - 2] * 0.25 + p[i * 2] * 0.5 + p[i * 2 + 2] * 0.25);
        o[i * 2 + 1] = r4(p[i * 2 - 1] * 0.25 + p[i * 2 + 1] * 0.5 + p[i * 2 + 3] * 0.25);
      }
      return o;
    }
    function hitText(s, q) {
      var L = textLayout(s, W), dx = q.x * W - L.cx, dy = q.y * W - L.cy, pd = W * 0.025;
      var cs = Math.cos(L.a), sn = Math.sin(L.a);
      var lx = dx * cs + dy * sn, ly = -dx * sn + dy * cs;   /* point in the text's own (unrotated) frame */
      return Math.abs(lx) <= L.w / 2 + pd && Math.abs(ly) <= L.h / 2 + pd;
    }
    function topTextAt(q) {
      for (var i = strokes.length - 1; i >= 0; i--) if (strokes[i].t === 'text' && hitText(strokes[i], q)) return i;
      return -1;
    }
    function settle(it) { var L = textLayout(it, W); it.x = r4(L.cx / W); it.y = r4(L.cy / W); }

    cv.addEventListener('pointerdown', function (e) {
      if (activeId !== null || (e.button && e.button !== 0)) return;
      e.preventDefault();
      rect = cv.getBoundingClientRect();
      var q = norm(e);
      if (tool === 'text') {
        if (float && float.editing) {
          if (!hitText(float.item, q)) { commitText(); return; }
          settle(float.item);
          drag = { kind: 'float', sx: q.x, sy: q.y, ox: float.item.x, oy: float.item.y, moved: false };
        } else {
          var idx = topTextAt(q);
          if (idx >= 0) {
            var o = strokes[idx];
            float = { item: { t: 'text', x: o.x, y: o.y, z: o.z, r: o.r || 0, text: o.text }, index: idx, snapshot: strokes.slice(), editing: false, moved: false };
            settle(float.item);
            strokes.splice(idx, 1); redrawBase();
            drag = { kind: 'item', sx: q.x, sy: q.y, ox: float.item.x, oy: float.item.y, moved: false };
          } else {
            drag = { kind: 'new', sx: q.x, sy: q.y, moved: false };
          }
        }
        activeId = e.pointerId;
        try { cv.setPointerCapture(e.pointerId); } catch (_) {}
        present();
        return;
      }
      if (float) commitText();
      activeId = e.pointerId;
      try { cv.setPointerCapture(e.pointerId); } catch (_) {}
      cur = { t: tool, w: CFG.sizes[sizeIdx], p: [], fx: q.x, fy: q.y };
      addPt(q, true); present();
    });
    cv.addEventListener('pointermove', function (e) {
      if (e.pointerId !== activeId) return;
      e.preventDefault();
      if (drag) {
        var q = norm(e), dx = q.x - drag.sx, dy = q.y - drag.sy;
        if (!drag.moved && dx * dx + dy * dy > 0.0001) drag.moved = true;
        if (drag.moved && drag.kind !== 'new') {
          float.item.x = drag.ox + dx; float.item.y = drag.oy + dy; float.moved = true;
          schedule();
        }
        return;
      }
      if (!cur) return;
      var list = e.getCoalescedEvents ? e.getCoalescedEvents() : [];
      if (!list || !list.length) list = [e];
      for (var i = 0; i < list.length; i++) addPt(norm(list[i]), false);
      schedule();
    });
    function endStroke(e) {
      activeId = null;
      if (!cur) return;
      if (e && typeof e.clientX === 'number') {
        var q = norm(e);
        if (cur.t !== 'erase' && CFG.smoothing > 0) {
          /* ease from the smoothed pen position to where the finger lifted, so loops close without a hook */
          var fx = cur.fx, fy = cur.fy;
          addPt({ x: fx + (q.x - fx) * 0.5, y: fy + (q.y - fy) * 0.5 }, true);
          addPt({ x: fx + (q.x - fx) * 0.85, y: fy + (q.y - fy) * 0.85 }, true);
        }
        addPt(q, true);
      }
      var s = cur; cur = null;
      delete s.fx; delete s.fy;
      if (s.t !== 'erase') s.p = relax(relax(s.p));
      history.push(strokes.slice());
      strokes.push(s);
      drawStroke(bctx, s, W); present();
      changed();
    }
    function endDrag(e, cancelled) {
      var d = drag; drag = null; activeId = null;
      if (d.kind === 'new') {
        if (!cancelled) openText({ t: 'text', x: d.sx, y: d.sy, z: textZ, r: 0, text: '' }, -1);
        else present();
        return;
      }
      settle(float.item);
      if (d.kind === 'float') { present(); if (!cancelled) focusText(); return; }
      if (d.moved) {
        strokes.splice(float.index, 0, float.item);
        history.push(float.snapshot);
        float = null; redrawBase(); present(); changed();
      } else if (cancelled) {
        strokes = float.snapshot; float = null; redrawBase(); present(); refresh();
      } else {
        float.editing = true; showBar();
      }
    }
    function onUp(cancelled) {
      return function (e) {
        if (e.pointerId !== activeId) return;
        if (drag) endDrag(e, cancelled);
        else endStroke(cancelled ? null : e);
      };
    }
    cv.addEventListener('pointerup', onUp(false));
    cv.addEventListener('pointercancel', onUp(true));
    cv.addEventListener('contextmenu', function (e) { e.preventDefault(); });

    function undo() {
      if (!history.length) return;
      strokes = history.pop(); redrawBase(); present(); changed();
    }
    function clearAll() {
      if (!strokes.length) return;
      history.push(strokes.slice()); strokes = []; redrawBase(); present(); changed();
    }

    /* ---------- text ---------- */
    function focusText() {
      try { textIn.focus({ preventScroll: true }); } catch (_) { textIn.focus(); }
      try { var n = textIn.value.length; textIn.setSelectionRange(n, n); } catch (_) {}
    }
    function showBar() {
      textbar.hidden = false;
      bDel.hidden = float.index < 0;
      textIn.value = float.item.text;
      slSize.value = float.item.z;
      slRot.value = float.item.r || 0;
      rotVal.textContent = (float.item.r || 0) + '\u00B0';
      present(); refresh();
      focusText();
    }
    function hideBar() {
      textbar.hidden = true;
      if (document.activeElement === textIn) textIn.blur();
    }
    function openText(item, idx) {
      float = { item: item, index: idx, snapshot: strokes.slice(), editing: true, moved: false };
      showBar();
    }
    function commitText() {
      if (!float) return;
      var f = float; float = null; hideBar();
      var it = f.item, txt = (it.text || '').replace(/\s+/g, ' ').replace(/^\s+|\s+$/g, ''), did = false;
      if (f.index >= 0) {
        var o = f.snapshot[f.index];
        if (!txt) did = true;
        else if (txt === o.text && it.z === o.z && (it.r || 0) === (o.r || 0) && !f.moved) strokes.splice(f.index, 0, o);
        else { it.text = txt; settle(it); strokes.splice(f.index, 0, it); did = true; }
      } else if (txt) {
        it.text = txt; settle(it); strokes.push(it); did = true;
      }
      if (did) history.push(f.snapshot);
      redrawBase(); present();
      if (did) changed(); else refresh();
    }
    function cancelText() {
      if (!float) return;
      strokes = float.snapshot; float = null; hideBar();
      redrawBase(); present(); refresh();
    }
    textIn.addEventListener('input', function () {
      if (!float) return;
      float.item.text = textIn.value; present(); refresh();
    });
    textIn.addEventListener('keydown', function (e) {
      if (e.isComposing) return;
      if (e.key === 'Enter' || e.keyCode === 13) { e.preventDefault(); e.stopPropagation(); commitText(); }
      else if (e.key === 'Escape' || e.key === 'Esc') { e.preventDefault(); e.stopPropagation(); cancelText(); }
    });
    function onSize() {
      if (!float) return;
      float.item.z = textZ = +slSize.value;
      float.moved = true;   /* marks the edit as a change */
      present(); refresh();
    }
    function onRot() {
      if (!float) return;
      var v = +slRot.value;
      if (Math.abs(v) <= 4) v = 0;                       /* snap to level */
      else if (Math.abs(Math.abs(v) - 90) <= 4) v = v > 0 ? 90 : -90;
      float.item.r = v; float.moved = true;
      rotVal.textContent = v + '\u00B0';
      present(); refresh();
    }
    slSize.addEventListener('input', onSize); slSize.addEventListener('change', onSize);
    slRot.addEventListener('input', onRot); slRot.addEventListener('change', onRot);
    [slSize, slRot].forEach(function (sl) {
      sl.addEventListener('keydown', function (e) { if (e.key === 'Enter' || e.keyCode === 13) { e.preventDefault(); e.stopPropagation(); commitText(); } });
    });
    textIn.addEventListener('keypress', function (e) {
      if (e.key === 'Enter' || e.keyCode === 13) { e.preventDefault(); e.stopPropagation(); }
    });

    function press(sel, btn) {
      var all = el.querySelectorAll(sel);
      for (var i = 0; i < all.length; i++) all[i].setAttribute('aria-pressed', all[i] === btn ? 'true' : 'false');
    }
    el.addEventListener('click', function (e) {
      var b = e.target.closest ? e.target.closest('button') : null;
      if (!b || !el.contains(b)) return;
      e.preventDefault();
      if (b.hasAttribute('data-tool')) {
        var nt = b.getAttribute('data-tool');
        if (float && nt !== 'text') commitText();
        tool = nt; press('[data-tool]', b);
        cv.style.cursor = tool === 'text' ? 'text' : 'crosshair';
        sizesGroup.hidden = tool === 'text';
        if (tool === 'text' && !float && statusEl.className.indexOf('warn') < 0) say('Tap the pad where you want words. Drag words to move them, tap them to edit, size, or rotate.', '');
      }
      else if (b.hasAttribute('data-size')) {
        sizeIdx = +b.getAttribute('data-size'); press('[data-size]', b);
      }
      else if (b.hasAttribute('data-body')) {
        var nb = b.getAttribute('data-body');
        if (nb === body) return;
        body = nb; press('[data-body]', b);
        syncSide();
        swapTemplate();
      }
      else if (b.hasAttribute('data-sex')) {
        var ns = b.getAttribute('data-sex');
        if (ns === sex) return;
        sex = ns; press('[data-sex]', b);
        if (body) swapTemplate();
      }
      else if (b.hasAttribute('data-side')) {
        var nd = b.getAttribute('data-side');
        if (!sideApplies() || nd === side) return;
        side = nd; press('[data-side]', b);
        swapTemplate();
      }
      else if (b === bUndo) { if (float) cancelText(); else undo(); }
      else if (b === bClear) { if (float) cancelText(); clearAll(); }
      else if (b === bDone) commitText();
      else if (b === bDel) { if (float) { float.item.text = ''; commitText(); } }
      else if (b === bAttach) { if (attachedName && !inkOrPending()) removeSketch(); else attach(); }
      else if (b === bPhoto) togglePhoto();
      else if (b === bSave) saveCopy();
    });
    el.addEventListener('keydown', function (e) {
      if (e.target === textIn) return;
      if ((e.metaKey || e.ctrlKey) && !e.shiftKey && (e.key === 'z' || e.key === 'Z')) { e.preventDefault(); if (float) cancelText(); else undo(); }
    });

    function swapTemplate() {
      buildUnder(); present();
      if (strokes.length || attachedName) changed(); else refresh();
      loadTemplate(fileKey(), function (k) {
        if (k !== fileKey()) return;
        buildUnder(); present(); refresh();
        if (FAILED[k] && statusEl.className.indexOf('warn') < 0) say('That outline didn\'t load. You can still sketch and describe the placement with Text.', '');
      });
    }
    function inkOrPending() { return hasInk() || !!(float && float.item.text.replace(/\s/g, '')); }

      var up = findUpload();
      if (up && attachedName) removeFromUpload(up.wrap, attachedName);
      attachedName = null; dirty = false;
      say('Sketch removed from your request.', '');
      refresh();
    }
    function setPhoto(open) {
      if (open) document.documentElement.classList.add('nsk-photo'); else document.documentElement.classList.remove('nsk-photo');
      bPhoto.textContent = open ? 'Hide the photo upload' : 'Have a placement photo instead? Upload it here';

      var open = !document.documentElement.classList.contains('nsk-photo');

    function hasInk() {
      for (var i = 0; i < strokes.length; i++) if (strokes[i].t !== 'erase') return true;
      return false;
    }
    function say(msg, cls) {
      statusEl.textContent = msg || '';
      statusEl.className = 'nsk-status' + (cls ? ' ' + cls : '');
    }
    function refresh() {
      if (attachedName && !busy) {
        /* the client may have deleted the sketch themselves from the photo box */
        var upc = findUpload();
        if (upc && !leafWithText(upc.wrap, attachedName)) { attachedName = null; dirty = false; }
      }
      var pending = !!(float && float.item.text.replace(/\s/g, ''));
      var ink = hasInk() || pending;
      var fk = fileKey(), loading = !!(fk && !READY[fk] && !FAILED[fk]);
      var showPh = loading || !(strokes.length || float || body);
      if (showPh) ph.innerHTML = loading ? '<b>Loading</b>' : '<b>Sketch here</b><span>Rough is perfect</span>';
      ph.style.opacity = showPh ? '1' : '0';
      bUndo.disabled = !history.length && !float;
      bClear.disabled = !strokes.length && !float;
      if (busy) return;
      if (!attachedName) { bAttach.textContent = 'Add sketch to my request'; bAttach.disabled = !ink; }
      else if (!ink) { bAttach.textContent = 'Remove sketch'; bAttach.disabled = false; }
      else if (dirty || pending) { bAttach.textContent = 'Update sketch'; bAttach.disabled = false; }
      else { bAttach.textContent = 'Sketch added'; bAttach.disabled = true; }
    }
    function changed() {
      if (attachedName) {
        dirty = true;
        if (hasInk()) say('Tap Update sketch to swap in this version.', '');
        else say('Your last sketch is still on your request. Tap Remove sketch to take it off.', '');
      }
      if (!bSave.hidden) exportFile().then(function (f) { savedFile = f; }, function () {});
      refresh();
    }

    /* ---------- export + attach ---------- */
    function nextName() {
      version++;
      return 'tattoo-sketch' + (body ? '-' + slugFor(body) + (inner() ? '-inner' : '') + (sex === 'm' ? '-male' : '') : '') + (version === 1 ? '' : '-' + version) + '.png';
    }
    function exportFile(name, noBody) {
      return new Promise(function (resolve, reject) {
        var n = CFG.exportPx, nh = Math.round(n * CFG.aspect), c = mk(n, nh), x = c.getContext('2d');
        x.fillStyle = CFG.paper; x.fillRect(0, 0, n, nh);
        if (!noBody) drawBody(x, n, nh, fileKey());
        var ink = mk(n, nh), ix = ink.getContext('2d');
        for (var i = 0; i < strokes.length; i++) drawStroke(ix, strokes[i], n);
        x.drawImage(ink, 0, 0);
        if (!c.toBlob) return reject(new Error('no toBlob'));
        try {
          c.toBlob(function (blob) {
            if (!blob) return reject(new Error('no blob'));
            try { resolve(new File([blob], name || 'tattoo-sketch.png', { type: 'image/png', lastModified: Date.now() })); }
            catch (err) { reject(err); }
          }, 'image/png');
        } catch (err) {
          if (!noBody && body) exportFile(name, true).then(resolve, reject); else reject(err);
        }
      });
    }
    function leafWithText(wrap, text) {
      var els = wrap.querySelectorAll('*');
      for (var i = 0; i < els.length; i++) {
        if (!els[i].children.length && (els[i].textContent || '').trim() === text) return els[i];
      }
      return null;
    }
    function removeFromUpload(wrap, name) {
      var n = leafWithText(wrap, name);
      while (n && n !== wrap && !n.querySelector('button')) n = n.parentElement;
      if (!n || n === wrap) return false;
      n.querySelector('button').click();
      return true;
    }
    function fileCount(wrap) {
      var c = wrap.querySelectorAll('.upload-card').length;
      if (c) return c;
      var m = (wrap.textContent || '').match(/(\d+)\s+files?\s+selected/i);
      return m ? +m[1] : 0;
    }
    function wait(ms) { return new Promise(function (r) { setTimeout(r, ms); }); }
    function until(test, ms) {
      var end = Date.now() + ms;
      return new Promise(function (resolve) {
        (function tick() {
          if (test()) return resolve(true);
          if (Date.now() > end) return resolve(false);
          setTimeout(tick, 120);
        })();
      });
    }

    function attach() {
      if (busy) return;
      if (float) commitText();
      if (!hasInk()) return;
      busy = true; bAttach.disabled = true; bAttach.textContent = 'Adding...';
      var up = findUpload();
      var name = nextName();
      exportFile(name).then(function (file) {
        savedFile = file;
        if (!up || typeof DataTransfer === 'undefined') { fallback(); return; }
        var mine = attachedName && leafWithText(up.wrap, attachedName);
        if (fileCount(up.wrap) - (mine ? 1 : 0) >= up.max) {
          if (up.placement) setPhoto(true);
          say('That upload is full (' + up.max + ' max). Remove a photo from it below, then try again.', 'warn');
          return;
        }
        if (mine) removeFromUpload(up.wrap, attachedName);
        return wait(mine ? 300 : 0).then(function () {
          var dt = new DataTransfer();
          dt.items.add(file);
          up.input.files = dt.files;
          up.input.dispatchEvent(new Event('change', { bubbles: true }));
          return until(function () { return !!leafWithText(up.wrap, name); }, 3500);
        }).then(function (ok) {
          if (!ok) { fallback(); return; }
          attachedName = name; dirty = false; bSave.hidden = true;
          say(up.placement ? 'Added to your request. Keep drawing and tap Update sketch to change it.'
                           : 'Added to Design Reference Photos below. Keep drawing and tap Update sketch to change it.', 'ok');
        });
      }).catch(function () { fallback(); }).then(function () { busy = false; refresh(); });
    }

    function fallback() {
      bSave.hidden = false;
      var up = findUpload();
      if (up && up.placement) setPhoto(true);
      say('Couldn\'t add it automatically. Tap Save sketch, then upload the saved image in the box below.', 'warn');
    }
    function saveCopy() {
      var file = savedFile;
      if (!file) return;
      var mobile = /iP(hone|ad|od)|Android/i.test(navigator.userAgent);
      if (mobile && navigator.canShare && navigator.canShare({ files: [file] })) {
        navigator.share({ files: [file] }).catch(function () {});
        return;
      }
      var url = URL.createObjectURL(file), a = document.createElement('a');
      a.href = url; a.download = file.name;
      document.body.appendChild(a); a.click(); a.remove();
      setTimeout(function () { URL.revokeObjectURL(url); }, 4000);
    }

    syncSide();
    refresh();
  }
})();
